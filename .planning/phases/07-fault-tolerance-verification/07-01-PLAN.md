Create an end-to-end verification test that demonstrates the complete fault tolerance loop:

1. **Setup**: Create a minimal test that:
   - Starts training with fault injection enabled
   - Injects a failure after first epoch
   - Verifies checkpoint is saved on exception
   - Verifies training can resume from checkpoint

2. **Test script** (`test_fault_tolerance.sh`):
```bash
#!/bin/bash
# End-to-end fault tolerance test
set -e

GPU=${1:-1}
EPOCHS=5
CKPT_EPOCH=2

echo "=== Fault Tolerance E2E Test (GPU=$GPU) ==="

# Clean start
rm -rf checkpoints/

# Run training with fault injection, expect failure at checkpoint epoch
echo "Step 1: Training with fault injection..."
timeout 120 torchrun --nproc_per_node=$GPU scripts/train_faceforensics.py \
  --epochs $EPOCHS \
  --checkpoint_interval $CKPT_EPOCH \
  --inject-fault \
  --inject-rank 0 \
  --inject-rate 0.5 \
  --limit 100 || true

# Verify checkpoint was saved
echo "Step 2: Checking checkpoint..."
if [ -f checkpoints/best_model.pt ]; then
    echo "  ✓ Checkpoint saved"
else
    echo "  ✗ No checkpoint found"
    exit 1
fi

# Resume training
echo "Step 3: Resuming from checkpoint..."
torchrun --nproc_per_node=$GPU scripts/train_faceforensics.py \
  --epochs $EPOCHS \
  --resume \
  --limit 100

echo "=== E2E Test Complete ==="
```
